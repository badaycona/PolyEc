cmake_minimum_required(VERSION 3.20)
project(PolyEc VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
enable_testing()
add_subdirectory(core)

option(POLYEC_BUILD_AI "Build CPU ONNX inference" OFF)
if(POLYEC_BUILD_AI)
  find_path(ORT_INCLUDE_DIR onnxruntime_cxx_api.h PATHS "${ORT_ROOT}/include" REQUIRED NO_DEFAULT_PATH)
  find_library(ORT_LIBRARY NAMES onnxruntime PATHS "${ORT_ROOT}/lib" REQUIRED NO_DEFAULT_PATH)
  add_library(polyec_ai STATIC native/policy.cpp native/lab.cpp native/realm_runtime.cpp)
  target_include_directories(polyec_ai PUBLIC native/include PRIVATE "${ORT_INCLUDE_DIR}")
  target_link_libraries(polyec_ai PUBLIC polyec_core "${ORT_LIBRARY}")
  add_executable(polyec_rl_replay native/replay.cpp)
  target_link_libraries(polyec_rl_replay PRIVATE polyec_ai)
  add_executable(polyec_realm_replay native/realm_replay.cpp)
  target_link_libraries(polyec_realm_replay PRIVATE polyec_ai)
  add_executable(polyec_realm_benchmark native/realm_benchmark.cpp)
  target_link_libraries(polyec_realm_benchmark PRIVATE polyec_ai)
endif()

option(POLYEC_BUILD_PYTHON "Build shared-core training binding" OFF)
if(POLYEC_BUILD_PYTHON)
  find_package(Python3 COMPONENTS Interpreter Development.Module REQUIRED)
  set(PYBIND11_FINDPYTHON ON)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(polyec_env MODULE NO_EXTRAS bindings/python/module.cpp bindings/python/realm_binding.cpp)
  target_link_libraries(polyec_env PRIVATE polyec_core)
  if(POLYEC_BUILD_AI)
    target_link_libraries(polyec_env PRIVATE polyec_ai)
    target_compile_definitions(polyec_env PRIVATE POLYEC_AI_ENABLED)
  endif()
endif()

option(POLYEC_BUILD_APP "Build native Godot extension" ON)
if(POLYEC_BUILD_APP)
  set(GODOTCPP_TARGET template_release CACHE STRING "" FORCE)
  set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/gdext/build_profile.json" CACHE PATH "" FORCE)
  add_subdirectory(third_party/godot-cpp)
  add_library(polyec SHARED gdext/polyec_session.cpp)
  target_link_libraries(polyec PRIVATE polyec_core godot-cpp)
  if(POLYEC_BUILD_AI)
    target_link_libraries(polyec PRIVATE polyec_ai)
    target_compile_definitions(polyec PRIVATE POLYEC_AI_ENABLED)
    add_custom_command(TARGET polyec POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${ORT_ROOT}/lib/onnxruntime.dll" "$<TARGET_FILE_DIR:polyec>/onnxruntime.dll"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${ORT_ROOT}/lib/onnxruntime_providers_shared.dll" "$<TARGET_FILE_DIR:polyec>/onnxruntime_providers_shared.dll")
  endif()
  target_compile_features(polyec PRIVATE cxx_std_20)
  target_compile_options(polyec PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
  set_target_properties(polyec PROPERTIES
    PREFIX ""
    OUTPUT_NAME "polyec.windows.template_release.x86_64"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/godot_project/bin"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/godot_project/bin")
  # Zig supplies libc++; godot-cpp's generic clang-on-Windows flags assume
  # a GNU libstdc++ install. Keep the upstream checkout untouched.
  if(CMAKE_CXX_COMPILER MATCHES "zig")
    set_property(TARGET godot-cpp PROPERTY INTERFACE_LINK_OPTIONS "-Wl,--no-undefined")
  endif()
  if(WIN32)
    add_executable(polyec_launcher WIN32 scripts/launcher.cpp)
    target_link_libraries(polyec_launcher PRIVATE user32)
    target_compile_options(polyec_launcher PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
    if(MINGW OR CMAKE_CXX_COMPILER MATCHES "zig")
      target_link_options(polyec_launcher PRIVATE -municode)
    endif()
    set_target_properties(polyec_launcher PROPERTIES OUTPUT_NAME "PolyEc")
  endif()
endif()
